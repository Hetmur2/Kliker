<body>

  <h1>Klikaj Piłkę!</h1>
  <div id="score">Punkty: 0</div>
  <br><br>
  <img id="ball" class="bouncing" src="https://cdn-icons-png.flaticon.com/512/951/951233.png" alt="Piłka" onclick="hitBall()">
  <div id="end-message"></div>

  <script>
    const ball = document.getElementById("ball");
    const scoreText = document.getElementById("score");
    const endMessage = document.getElementById("end-message");

    let score = 0;
    const maxScore = 500;

    const ballImages = [
      "https://cdn-icons-png.flaticon.com/512/951/951233.png",
      "https://cdn-icons-png.flaticon.com/512/3429/3429872.png",
      "https://cdn-icons-png.flaticon.com/512/2795/2795260.png",
      "https://cdn-icons-png.flaticon.com/512/833/833314.png",
      "https://cdn-icons-png.flaticon.com/512/7185/7185565.png"
    ];

    function hitBall() {
      if (score >= maxScore) return;

      score++;
      scoreText.textContent = `Punkty: ${score}`;

      // Efekt kliknięcia
      ball.style.animation = "pop 0.2s";
      setTimeout(() => ball.style.animation = "bounce 1s infinite cubic-bezier(0.68, -0.55, 0.27, 1.55)", 200);

      // Zmiana wyglądu co 20 kliknięć
      if (score % 20 === 0) {
        const index = (score / 20) % ballImages.length;
        ball.src = ballImages[index];
        ball.classList.add("animate");
        setTimeout(() => ball.classList.remove("animate"), 400);
      }

      // Zakończenie gry
      if (score === maxScore) {
        endMessage.textContent = "🎉 Gratulacje! Zdobyłeś maksymalną liczbę podbić!";
        ball.style.pointerEvents = "none";
        ball.style.filter = "grayscale(1)";
        ball.style.boxShadow = "none";
      }
    }
  </script>
